PARTE 3 - JAVASCRIPT

// Função responsável por calcular o consumo médio de combustível

function calcular() {
                      // Pegando os valores digitados pelo usuário e convertendo para número decimal

                       const km = parseFloat(document.getElementById('km').value);
                       const litros = parseFloat(document.getElementById('litros').value);
                       const resultado = document.getElementById('resultado');

                      // Validação simples: se os valores forem inválidos ou zero, mostra mensagem de erro

                        if (!km || !litros || litros === 0) {  
                            resultado.innerText = 'Por favor, insira valores válidos.';
                            return; - // Interrompe a função aqui
                                                            }

                        // Faz o cálculo dividindo a distância pelos litros abastecidos

                        const consumo = km / litros;

                        // Exibe o resultado formatado com 2 casas decimais

                        resultado.innerText = `Consumo médio: ${consumo.toFixed(2)} km/l`;
                    }
